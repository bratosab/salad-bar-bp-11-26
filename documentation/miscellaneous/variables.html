src/app/salad/store/salad.effects.ts

getToppingsEffect
Type : unknown
Default value : createEffect( (actions$ = inject(Actions), saladService = inject(SaladService)) => { return actions$.pipe( ofType(GetToppings), exhaustMap(() => saladService.getToppings().pipe( map((toppings) => SaveToppings({ list: toppings })) ), ), ); }, { functional: true }, )

src/app/salad/store/salad.reducer.ts

Type : SaladState
Default value : { toppings: [], chosenToppings: [], }
saladReducer
Type : unknown
Default value : createReducer( initialState, on(SaveToppings, (state, action) => { return { ...state, toppings: action.list }; }), on(ChooseTopping, (state, action) => { return { ...state, chosenToppings: [...state.chosenToppings, action.toppingToAdd] }; }), on(RemoveTopping, (state, action) => { const filteredToppings = state.chosenToppings.filter( (topping) => topping.id !== action.toppingToRemove.id, ); return { ...state, chosenToppings: [...filteredToppings] }; }), )
